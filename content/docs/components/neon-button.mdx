## Usage

```tsx
import { NeonButton } from "@/components/pioneerui/neon-button"

export default function Page() {
  return (
    <div className="bg-zinc-950 p-8">
      <NeonButton color="purple">Launch App</NeonButton>
      <NeonButton color="cyan" size="lg">Connect Wallet</NeonButton>
    </div>
  )
}
```

## Props

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `color` | `"purple" \| "cyan" \| "green" \| "pink" \| "orange"` | `"purple"` | Neon glow color theme |
| `size` | `"sm" \| "md" \| "lg"` | `"md"` | Button size |
| `children` | `ReactNode` | — | Button label |
